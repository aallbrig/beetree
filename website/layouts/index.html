A tree is just readable YAML

Write it by hand, build it in the browser, or generate it from the CLI. Every tool speaks the same format.

Read the format guide

version: "1.0"
metadata:
  name: guard
tree:
  type: selector
  name: guard_ai
  children:
    - type: sequence
      name: combat
      children:
        - {type: condition, name: can_see_target}
        - {type: action, name: chase_target}
    - {type: action, name: patrol}
